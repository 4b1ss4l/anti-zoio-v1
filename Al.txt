            let isProcessing = false;
            let isMinimized = false;
            let isConfigOpen = false;

            // ============================================
            // BANCO DE CONHECIMENTO LOCAL (FALLBACK INTELIGENTE)
            // ============================================
            const KNOWLEDGE = {
                'oi|olá|opa|eai': 'Olá! Como posso ajudar você hoje? 😊',
                'tudo bem|como vai|beleza': 'Estou bem, obrigada! E você, como está?',
                'seu nome|quem é você': 'Sou uma assistente IA criada para ajudar você com o que precisar!',
                'obrigado|valeu|agradeço': 'Por nada! Fico feliz em ajudar. 😊',
                'tchau|adeus|até logo': 'Até mais! Foi um prazer ajudar. 👋',
                'ajuda|o que faz|pode fazer': 'Posso: responder perguntas, programar, traduzir, calcular, pesquisar, dar dicas, explicar conceitos e muito mais!',
                'matemática|calcular|quanto é': 'Posso fazer cálculos! Me diga a operação com +, -, *, / ou porcentagem.',
                'programação|código|javascript|python': 'Sou ótima em programação! Me diga o que precisa que eu ajudo com código.',
                'traduzir|tradução|inglês|português': 'Posso traduzir! Diga a frase e o idioma de destino.',
                'significa|definição|conceito|o que é': 'Vou pesquisar isso para você! Me dê mais contexto.',

                'poema|poesia|verso': '🌹 "Na imensidão do código, encontrei um novo mundo. Cada linha uma descoberta, cada erro um aprendizado."',

            }

            // ============================================
            // MENSAGENS
            // ============================================

            function addMessage(text, type, extra = {}) {
                const div = document.createElement('div');
                div.className = `message ${type}`;

                if (type === 'ai' && extra.think) {
                    div.innerHTML = `
                        <div class="think">💭 ${extra.think}</div>
                        ${text}
                    `;
                } else if (type === 'ai' && extra.code) {
                    div.innerHTML = `
                        <div class="code-block">${escapeHtml(text)}</div>
                    `;
                } else {
                    div.textContent = text;
                }

                DOM.messages.appendChild(div);
                scrollToBottom();
                return div;
            }

            function escapeHtml(text) {
                const div = document.createElement('div');
                div.textContent = text;
                return div.innerHTML;
            }

            function showTyping() {
                const div = document.createElement('div');
                div.className = 'message ai';
                div.id = 'typing-indicator';
                div.innerHTML = `
                    <div class="typing-dots">
                        <span></span>
                        <span></span>
                        <span></span>
                    </div>
                `;
                DOM.messages.appendChild(div);
                scrollToBottom();
            }

            function removeTyping() {
                const typing = document.getElementById('typing-indicator');
                if (typing) typing.remove();
            }

            // ============================================
            // PROCESSAMENTO INTELIGENTE
            // ============================================

            async function sendMessage() {
                const text = DOM.input.value.trim();
                if (!text || isProcessing) return;

                // Adiciona mensagem do usuário
                addMessage(text, 'user');
                DOM.input.value = '';
                DOM.send.disabled = true;
                autoResizeInput();

                isProcessing = true;
                showTyping();

                try {
                    let resposta, think = '';

                    // Tenta usar API se configurada
                    if (CONFIG.apiKey && CONFIG.apiType !== 'local') {
                        try {
                            const apiResponse = await queryAPI(text);
                            if (apiResponse) {
                                resposta = apiResponse.answer;
                                think = apiResponse.think || 'Pensando...';
                            } else {
                                throw new Error('API não respondeu');
                            }
                        } catch (apiError) {
                            console.warn('API falhou, usando fallback:', apiError);
                            const local = getLocalAnswer(text);
                            if (local) {
                                resposta = local;
                                think = 'Usando conhecimento local';
                            } else {
                                resposta = generateSmartResponse(text);
                                think = 'Gerando resposta inteligente...';
                            }
                        }
                    } else {
                        // Fallback local
                        const local = getLocalAnswer(text);
                        if (local) {
                            resposta = local;
                            think = 'Conhecimento local';
                        } else {
                            resposta = generateSmartResponse(text);
                            think = 'Gerando resposta...';
                        }
                    }

                    await sleep(300 + Math.random() * 400);
                    removeTyping();

                    // Detecta se é código
                    const isCode = resposta.includes('```') || 
                                  resposta.includes('function') || 
                                  resposta.includes('class ') ||
                                  resposta.includes('import ');

                    addMessage(resposta, 'ai', { think, code: isCode });

                } catch (error) {
                    console.error('Erro:', error);
                    removeTyping();
                    addMessage('❌ Desculpe, tive um problema. Tente novamente ou configure uma API válida.', 'ai');
                }

                isProcessing = false;
                DOM.send.disabled = false;
                DOM.input.focus();
            }

            // ============================================
            // RESPOSTA LOCAL INTELIGENTE
            // ============================================

            function getLocalAnswer(question) {
                const lower = question.toLowerCase().trim();

                // Verifica padrões
                for (const [pattern, response] of Object.entries(KNOWLEDGE)) {
                    const patterns = pattern.split('|');
                    for (const p of patterns) {
                        if (lower.includes(p)) {
                            return response;
                        }
                    }
                }

                // Matemática
                if (lower.match(/[\d\+\-\*\/\%]/)) {
                    return solveMath(question);
                }

                // Data/Hora
                if (lower.includes('hora') || lower.includes('horário')) {
                    return `🕐 Agora são ${new Date().toLocaleTimeString('pt-BR')}`;
                }

                if (lower.includes('data') || lower.includes('hoje')) {
                    return `📅 Hoje é ${new Date().toLocaleDateString('pt-BR', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' })}`;
                }

                // Clima (simulado)
                if (lower.includes('clima') || lower.includes('tempo') || lower.includes('temperatura')) {
                    return '🌤️ Para informações climáticas precisas, recomendo consultar um site especializado como o Climatempo.';
                }

                // Tradução simples
                if (lower.includes('traduzir') && lower.includes('inglês')) {
                    return '🔤 Diga a palavra ou frase em português que quer traduzir para inglês.';
                }

                return null;
            }

            // ============================================
            // RESOLVER MATEMÁTICA
            // ============================================

            function solveMath(question) {
                try {
                    // Limpa a pergunta
                    const clean = question
                        .replace(/quanto é/gi, '')
                        .replace(/calcular/gi, '')
                        .replace(/qual é/gi, '')
                        .replace(/quanto dá/gi, '')
                        .replace(/=/g, '')
                        .trim();

                    // Extrai números e operadores
                    const match = clean.match(/([\d.,]+)\s*([+\-*/%])\s*([\d.,]+)/);
                    if (!match) {
                        return '🧮 Por favor, diga a operação claramente. Exemplo: "2 + 2" ou "10 * 5"';
                    }

                    const num1 = parseFloat(match[1].replace(',', '.'));
                    const operator = match[2];
                    const num2 = parseFloat(match[3].replace(',', '.'));

                    if (isNaN(num1) || isNaN(num2)) {
                        return '🧮 Não entendi os números. Tente algo como "5 + 3"';
                    }

                    let result;
                    let operation;

                    switch (operator) {
                        case '+':
                            result = num1 + num2;
                            operation = 'soma';
                            break;
                        case '-':
                            result = num1 - num2;
                            operation = 'subtração';
                            break;
                        case '*':
                        case 'x':
                            result = num1 * num2;
                            operation = 'multiplicação';
                            break;
                        case '/':
                        case '÷':
                            if (num2 === 0) return '❌ Não posso dividir por zero!';
                            result = num1 / num2;
                            operation = 'divisão';
                            break;
                        case '%':
                            result = (num1 * num2) / 100;
                            operation = 'porcentagem';
                            break;
                        default:
                            return '🧮 Operador não reconhecido. Use +, -, *, / ou %';
                    }

                    return `🧮 Resultado da ${operation}: **${num1} ${operator} ${num2} = ${result}**`;
                } catch (e) {
                    return '🧮 Não entendi a operação. Tente: "2 + 2" ou "10 * 5"';
                }
            }

            function generateSmartResponse(question) {
                const lower = question.toLowerCase();
                const templates = [
                    `🤔 Interessante pergunta sobre "${question.substring(0, 30)}..."`,
                    `📝 Vou analisar isso: "${question}"`,
                    `💡 Deixe-me pensar sobre "${question}"...`,
                    `🔍 Ótima pergunta! Sobre isso:`,
                    `✨ Que pergunta interessante!`
                ];

                let response = templates[Math.floor(Math.random() * templates.length)];

                // Adiciona contexto baseado no tipo de pergunta
                if (lower.includes('como') || lower.includes('qual')) {
                    response += '\n\n💡 Para responder melhor, você pode configurar uma API (ícone ⚙️) ou me dar mais detalhes.';
                } else if (lower.includes('por que') || lower.includes('porque')) {
                    response += '\n\n🔬 Essa é uma pergunta complexa. Com uma API configurada, posso dar uma resposta mais precisa!';
                } else if (lower.length < 20) {
                    response += '\n\n🤷 Pode reformular a pergunta? Quanto mais detalhes, melhor posso ajudar.';
                } else {
                    response += '\n\n📚 Usei meu conhecimento local. Para respostas mais precisas, configure uma API.';
                }

                return response;
            }

            // API INTEGRAÇÃO (MULTIPLATAFORMA)
            // ============================================

            async function queryAPI(question) {
                const config = {
                    openai: {
                        url: 'https://api.openai.com/v1/chat/completions',
                        headers: {
                            'Authorization': `Bearer ${CONFIG.apiKey}`,
                            'Content-Type': 'application/json'
                        },
                        body: {
                            model: CONFIG.model,
                            messages: [
                                { role: 'system', content: 'Você é uma assistente IA útil, amigável e inteligente. Responda de forma clara e concisa.' },
                                { role: 'user', content: question }
                            ],
                            max_tokens: CONFIG.maxTokens,
                            temperature: CONFIG.temperature
                        }
                    },
                    anthropic: {
                        url: 'https://api.anthropic.com/v1/messages',
                        headers: {
                            'x-api-key': CONFIG.apiKey,
                            'anthropic-version': '2023-06-01',
                            'Content-Type': 'application/json'
                        },
                        body: {
                            model: CONFIG.model,
                            max_tokens: CONFIG.maxTokens,
                            messages: [{ role: 'user', content: question }]
                        }
                    },
                    google: {
                        url: 'https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: {
                            contents: [{ parts: [{ text: question }] }]
                        }
                    }
                };

                const api = config[CONFIG.apiType];
                if (!api) throw new Error('Tipo de API não suportado');

                // Adiciona key para Google
                let url = api.url;
                if (CONFIG.apiType === 'google') {
                    url += `?key=${CONFIG.apiKey}`;
                }

                const response = await fetch(url, {
                    method: 'POST',
                    headers: api.headers,
                    body: JSON.stringify(api.body)
                });

                if (!response.ok) {
                    const error = await response.text();
                    throw new Error(`API Error: ${response.status} - ${error}`);
                }

                const data = await response.json();

                // Processa resposta dependendo da API
                let answer = '';
                let think = '';

                switch (CONFIG.apiType) {
                    case 'openai':
                        answer = data.choices?.[0]?.message?.content || '';
                        break;
                    case 'anthropic':
                        answer = data.content?.[0]?.text || '';
                        break;
                    case 'google':
                        answer = data.candidates?.[0]?.content?.parts?.[0]?.text || '';
                        break;
                }

                if (!answer) {
                    throw new Error('Resposta vazia da API');
                }

                return { answer, think: `Usando ${CONFIG.apiType} (${CONFIG.model})` };
            }
